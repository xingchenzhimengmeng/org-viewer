from dataclasses import dataclass
import asyncio

# Retry a flaky request with exponential backoff
@dataclass
class Client:
    base_url: str
    timeout: float = 3.5

    async def fetch(self, path: str, retries: int = 3) -> dict:
        """Return decoded JSON for *path*."""
        for attempt in range(retries):
            try:
                return await self._get(path)
            except TimeoutError as exc:
                delay = 0.25 * (2 ** attempt)
                print(f"retry {attempt}: {exc}")
                await asyncio.sleep(delay)
        raise RuntimeError("gave up")

export async function loadThemes(url: string): Promise<Theme[]> {
  const response = await fetch(url);           // network call
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  const themes = (await response.json()) as Theme[];
  return themes.filter((t) => t.name.length > 0);
}

public record Point(int x, int y) {
    @Override
    public String toString() {
        return String.format("(%d, %d)", x, y);
    }

    public static List<Point> parse(String[] rows) {
        List<Point> out = new ArrayList<>();
        for (String row : rows) {
            String[] parts = row.split(",");
            out.add(new Point(Integer.parseInt(parts[0]), Integer.parseInt(parts[1])));
        }
        return out;
    }
}

use std::collections::HashMap;

#[derive(Debug, Clone)]
struct Stats { hits: u64 }

fn tally(keys: &[&str]) -> HashMap<&str, Stats> {
    let mut map = HashMap::new();
    for key in keys {
        let entry = map.entry(*key).or_insert(Stats { hits: 0 });
        entry.hits += 1;
    }
    map
}

FROM orders o
JOIN customers c ON c.id = o.customer_id
WHERE o.created_at >= '2026-01-01'
GROUP BY c.name
HAVING SUM(o.amount) > 1000
ORDER BY revenue DESC
